# ================================================================ # GAMBIA CENSUS POINT MAPPING AND SPATIAL ANALYSIS # CensusPoints.csv + Census_EAs_Geo.json # # Produced by Ansu Singhateh # GIS Analyst | GIS / Cartography Unit | Gambia Bureau of Statistics # GBoS GEOAI • Research and educational use # ================================================================ # 1. INSTALL AND LOAD PACKAGES ------------------------------------ packages <- c( "sf", "dplyr", "ggplot2", "readr", "stringr", "janitor", "tidyr", "leaflet", "viridis", "scales", "htmlwidgets" ) missing <- packages[!packages %in% installed.packages()[, "Package"]] if (length(missing) > 0) { install.packages(missing) } invisible(lapply(packages, library, character.only = TRUE)) # 2. SELECT INPUT FILES -------------------------------------------- cat("Select CensusPoints.csv\n") points_file <- file.choose() cat("Select Census_EAs_Geo.json\n") ea_file <- file.choose() output_dir <- file.path(getwd(), "Gambia_Census_Analysis") dir.create(output_dir, showWarnings = FALSE, recursive = TRUE) # 3. READ DATA ----------------------------------------------------- points_raw <- read_csv( points_file, show_col_types = FALSE ) %>% clean_names() ea <- st_read( ea_file, quiet = TRUE ) %>% clean_names() cat("\nPOINT DATA STRUCTURE\n") print(names(points_raw)) cat("\nEA BOUNDARY STRUCTURE\n") print(names(ea)) cat("\nPOINT RECORDS:", nrow(points_raw)) cat("\nEA FEATURES:", nrow(ea), "\n") # 4. AUTOMATIC FIELD DETECTION ------------------------------------- find_field <- function(data, candidates) { result <- intersect(candidates, names(data)) if (length(result) == 0) return(NA_character_) result[1] } lon_field <- find_field( points_raw, c("longitude", "lon", "long", "x", "gps_longitude") ) lat_field <- find_field( points_raw, c("latitude", "lat", "y", "gps_latitude") ) # IMPORTANT: # If your coordinate columns have different names, # specify them manually here. # # Example: # lon_field <- "gps2" # lat_field <- "gps" if (is.na(lon_field) || is.na(lat_field)) { stop( paste( "Longitude/latitude fields were not detected.", "Inspect names(points_raw) and specify lon_field and lat_field." ) ) } cat("\nLongitude field:", lon_field) cat("\nLatitude field:", lat_field, "\n") # 5. CLEAN COORDINATES --------------------------------------------- points_raw <- points_raw %>% mutate( longitude = parse_number(as.character(.data[[lon_field]])), latitude = parse_number(as.character(.data[[lat_field]])) ) # Gambia geographic coordinate validation points_raw <- points_raw %>% mutate( valid_coordinates = !is.na(longitude) & !is.na(latitude) & between(longitude, -18, -12) & between(latitude, 12, 14) ) cat("\nValid coordinates:", sum(points_raw$valid_coordinates), "\n") cat("Invalid coordinates:", sum(!points_raw$valid_coordinates), "\n") points_clean <- points_raw %>% filter(valid_coordinates) # WGS84 geographic reference system points_sf <- st_as_sf( points_clean, coords = c("longitude", "latitude"), crs = 4326, remove = FALSE ) # Ensure EA boundaries use WGS84 if (is.na(st_crs(ea))) { stop("EA boundary CRS is missing. Assign its verified source CRS first.") } ea <- st_transform(ea, 4326) # 6. DETECT EA ADMINISTRATIVE FIELDS ------------------------------- region_field <- find_field( ea, c("rco", "region", "region_name", "lganame", "lga_name") ) district_field <- find_field( ea, c("distname", "district", "district_name") ) ward_field <- find_field( ea, c("wardname", "ward", "ward_name") ) ea_field <- find_field( ea, c("eaname", "ea_name", "feacode", "ea_code", "ea", "ea_code") ) cat("\nDetected EA fields:\n") cat("Region:", region_field, "\n") cat("District:", district_field, "\n") cat("Ward:", ward_field, "\n") cat("EA:", ea_field, "\n") # Create standard administrative fields get_attribute <- function(data, field) { if (is.na(field)) { return(rep(NA_character_, nrow(data))) } as.character(data[[field]]) } ea$region_name <- get_attribute(ea, region_field) ea$district_name <- get_attribute(ea, district_field) ea$ward_name <- get_attribute(ea, ward_field) ea$ea_identifier <- get_attribute(ea, ea_field) # 7. SPATIAL POINT-IN-POLYGON ANALYSIS ----------------------------- cat("\nAssigning census points to Enumeration Areas...\n") # Retain original point attributes and attach EA attributes point_assignment <- st_join( points_sf, ea %>% select( region_name, district_name, ward_name, ea_identifier ), join = st_within, left = TRUE ) # Check unassigned records unassigned <- point_assignment %>% filter(is.na(ea_identifier)) assigned <- point_assignment %>% filter(!is.na(ea_identifier)) cat("\nAssigned points:", nrow(assigned)) cat("\nUnassigned points:", nrow(unassigned), "\n") # 8. POPULATION / HOUSEHOLD / STRUCTURE VARIABLES ------------------ pop_field <- find_field( points_raw, c("pop", "population", "total_population", "sum_pop") ) hh_field <- find_field( points_raw, c("hhs", "households", "hh", "num_households", "sum_hhs") ) structure_field <- find_field( points_raw, c("numstructu", "structures", "num_structures", "structure_count") ) # Safe numerical extraction get_numeric <- function(data, field) { if (is.na(field)) { return(rep(NA_real_, nrow(data))) } suppressWarnings( parse_number(as.character(data[[field]])) ) } assigned$population_value <- get_numeric(assigned, pop_field) assigned$household_value <- get_numeric(assigned, hh_field) assigned$structure_value <- get_numeric(assigned, structure_field) # 9. ENUMERATION AREA SUMMARY -------------------------------------- ea_summary <- assigned %>% st_drop_geometry() %>% group_by( region_name, district_name, ward_name, ea_identifier ) %>% summarise( census_points = n(), population = if (all(is.na(population_value))) { NA_real_ } else { sum(population_value, na.rm = TRUE) }, households = if (all(is.na(household_value))) { NA_real_ } else { sum(household_value, na.rm = TRUE) }, structures = if (all(is.na(structure_value))) { NA_real_ } else { sum(structure_value, na.rm = TRUE) }, .groups = "drop" ) # 10. REGIONAL SUMMARY --------------------------------------------- regional_summary <- ea_summary %>% group_by(region_name) %>% summarise( enumeration_areas = n_distinct(ea_identifier), census_points = sum(census_points, na.rm = TRUE), population = if (all(is.na(population))) { NA_real_ } else { sum(population, na.rm = TRUE) }, households = if (all(is.na(households))) { NA_real_ } else { sum(households, na.rm = TRUE) }, structures = if (all(is.na(structures))) { NA_real_ } else { sum(structures, na.rm = TRUE) }, .groups = "drop" ) # 11. DISTRICT SUMMARY --------------------------------------------- district_summary <- ea_summary %>% group_by(region_name, district_name) %>% summarise( enumeration_areas = n_distinct(ea_identifier), census_points = sum(census_points, na.rm = TRUE), population = if (all(is.na(population))) { NA_real_ } else { sum(population, na.rm = TRUE) }, households = if (all(is.na(households))) { NA_real_ } else { sum(households, na.rm = TRUE) }, structures = if (all(is.na(structures))) { NA_real_ } else { sum(structures, na.rm = TRUE) }, .groups = "drop" ) # 12. WARD SUMMARY -------------------------------------------------- ward_summary <- ea_summary %>% group_by(region_name, district_name, ward_name) %>% summarise( enumeration_areas = n_distinct(ea_identifier), census_points = sum(census_points, na.rm = TRUE), population = if (all(is.na(population))) { NA_real_ } else { sum(population, na.rm = TRUE) }, households = if (all(is.na(households))) { NA_real_ } else { sum(households, na.rm = TRUE) }, .groups = "drop" ) # 13. PROJECTED EA AREA AND POINT DENSITY --------------------------- # UTM Zone 28N, appropriate for The Gambia ea_projected <- st_transform(ea, 32628) ea_area <- ea_projected %>% mutate( area_km2 = as.numeric(st_area(geometry)) / 1e6 ) %>% st_drop_geometry() %>% select(ea_identifier, area_km2) ea_summary <- ea_summary %>% left_join(ea_area, by = "ea_identifier") %>% mutate( point_density_km2 = ifelse( area_km2 > 0, census_points / area_km2, NA_real_ ) ) # 14. MAP 1: CENSUS POINT DISTRIBUTION ----------------------------- map_points <- ggplot() + geom_sf( data = ea, fill = "grey97", colour = "grey65", linewidth = 0.12 ) + geom_sf( data = points_sf, colour = "#D73027", size = 0.35, alpha = 0.65 ) + labs( title = "Gambia Census Point Distribution", subtitle = "Census points over Enumeration Area boundaries", caption = "Produced by Ansu Singhateh | GIS / Cartography Unit | GBoS", x = "Longitude", y = "Latitude" ) + theme_minimal() ggsave( file.path(output_dir, "01_Census_Point_Distribution.png"), map_points, width = 12, height = 9, dpi = 300 ) # 15. MAP 2: POINT DENSITY BY EA ----------------------------------- ea_map_data <- ea %>% left_join( ea_summary %>% select(ea_identifier, census_points, point_density_km2), by = "ea_identifier" ) map_density <- ggplot(ea_map_data) + geom_sf( aes(fill = point_density_km2), colour = "white", linewidth = 0.08 ) + scale_fill_viridis_c( option = "magma", na.value = "grey85", name = "Points/km²" ) + labs( title = "Census Point Density by Enumeration Area", subtitle = "Number of mapped census points per square kilometre", caption = "GBoS GIS / Cartography Unit", x = "Longitude", y = "Latitude" ) + theme_minimal() ggsave( file.path(output_dir, "02_Census_Point_Density.png"), map_density, width = 12, height = 9, dpi = 300 ) # 16. CHART: REGIONAL CENSUS POINTS ------------------------------- chart_region <- ggplot( regional_summary, aes( x = reorder(region_name, census_points), y = census_points ) ) + geom_col(fill = "#2471A3") + coord_flip() + labs( title = "Census Points by Region", x = "Region", y = "Number of census points" ) + theme_minimal() ggsave( file.path(output_dir, "03_Regional_Census_Points.png"), chart_region, width = 10, height = 7, dpi = 300 ) # 17. LEAFLET INTERACTIVE MAP -------------------------------------- interactive_map <- leaflet() %>% addProviderTiles("Esri.WorldImagery") %>% addPolygons( data = ea, color = "white", weight = 0.5, fillOpacity = 0.08, popup = ~paste( "EA:", ea_identifier, "
Region:", region_name, "
District:", district_name, "
Ward:", ward_name ) ) %>% addCircleMarkers( data = points_sf, radius = 2, color = "#FF4500", stroke = FALSE, fillOpacity = 0.7, popup = ~paste( "Population:", if (!is.na(pop_field)) { as.character(.data[[pop_field]]) } else { "Not available" } ) ) %>% addLayersControl( overlayGroups = c("Census Points", "Enumeration Areas"), options = layersControlOptions(collapsed = FALSE) ) htmlwidgets::saveWidget( interactive_map, file.path(output_dir, "04_Interactive_Census_Map.html"), selfcontained = FALSE ) # 18. EXPORT RESULTS ----------------------------------------------- write_csv( ea_summary, file.path(output_dir, "EA_Census_Summary.csv") ) write_csv( regional_summary, file.path(output_dir, "Regional_Census_Summary.csv") ) write_csv( district_summary, file.path(output_dir, "District_Census_Summary.csv") ) write_csv( ward_summary, file.path(output_dir, "Ward_Census_Summary.csv") ) st_write( assigned, file.path(output_dir, "Assigned_Census_Points.gpkg"), delete_dsn = TRUE, quiet = TRUE ) st_write( unassigned, file.path(output_dir, "Unassigned_Census_Points.gpkg"), delete_dsn = TRUE, quiet = TRUE ) st_write( ea_map_data, file.path(output_dir, "EA_Point_Density.gpkg"), delete_dsn = TRUE, quiet = TRUE ) # 19. FINAL ANALYTICAL REPORT -------------------------------------- cat("\n========================================\n") cat("GAMBIA CENSUS POINT ANALYSIS COMPLETE\n") cat("========================================\n") cat("\nTotal source points:", nrow(points_raw)) cat("\nValid points:", nrow(points_sf)) cat("\nInvalid coordinates:", sum(!points_raw$valid_coordinates)) cat("\nAssigned points:", nrow(assigned)) cat("\nUnassigned points:", nrow(unassigned)) cat("\nEA boundary features:", nrow(ea)) cat("\nRegions:", n_distinct(ea$region_name)) cat("\nDistricts:", n_distinct(ea$district_name)) cat("\nWards:", n_distinct(ea$ward_name)) cat("\n\nOUTPUT DIRECTORY:\n") cat(normalizePath(output_dir), "\n") cat("\nAnalysis completed successfully.\n")
